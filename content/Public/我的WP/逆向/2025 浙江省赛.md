void RC4_KSA(unsigned char *key, int keyLength, unsigned char *S) {
    int i, j = 0;
    unsigned char temp;

        S[i] = i;

        S[i] = S[j];
        S[j] = temp;

}

// RC4伪随机生成算法 (PRGA)
void RC4_PRGA(unsigned char *S, unsigned char *data, int dataLength) {
    int i = 0, j = 0, k;
    unsigned char temp;
    
    for (k = 0; k < dataLength; k++) {
        i = (i + 1) % 256;

        data[k] ^= S[(S[i] + S[j]) % 256];

def KSA(key):
    """ KSA 密钥拓展 """
    """ 用来扩散密钥 """
    S = list(range(256))
    tmp = 0
    for i in range(256):
        tmp = (tmp + S[i] + key[i % len(key)]) % 256
        S[i], S[tmp] = S[tmp], S[i]
    return S
 
def PRGA(S):
    """ PRGA 伪随机数生成算法 """
    i, j = 0, 0
    while True:
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        K = S[(S[i] + S[j]) % 256]
        yield K
 
def RC4(key, text):
    """ RC4 加密流程 """
    S = KSA(key)
    keystream = PRGA(S)
    res = []
    for char in text:
        res.append(char ^ next(keystream))
    return bytes(res)
